   106→function formatUSD(amount: number): string {
   107→  return new Intl.NumberFormat('en-US', {
   108→    style: 'currency',
   109→    currency: 'USD',
   110→    minimumFractionDigits: 0,
   111→    maximumFractionDigits: 0,
   112→  }).format(amount)
   113→}

   115→function formatShortUSD(amount: number): string {
   116→  if (amount >= 1_000_000_000) return `$${(amount / 1_000_000_000).toFixed(1)}B`
   117→  if (amount >= 1_000_000) return `$${(amount / 1_000_000).toFixed(1)}M`
   118→  if (amount >= 1_000) return `$${(amount / 1_000).toFixed(0)}K`
   119→  return `$${amount.toFixed(0)}`
   120→}

   122→// ─── CSV Export Utility ─────────────────────────────────────────────────────
   123→function exportToCSV(data: Record<string, unknown>[], filename: string) {
   124→  if (data.length === 0) return
   125→  const headers = Object.keys(data[0])
   126→  const csv = [
   127→    headers.join(','),
   128→    ...data.map(row => headers.map(h => {
   129→      const val = String(row[h] ?? '')
   130→      return val.includes(',') || val.includes('"') || val.includes('\n') ? `"${val.replace(/"/g, '""')}"` : val
   131→    }).join(','))
   132→  ].join('\n')
   133→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   134→  const url = URL.createObjectURL(blob)
   135→  const a = document.createElement('a')
   136→  a.href = url
   137→  a.download = filename
   138→  a.click()
   139→  URL.revokeObjectURL(url)
   140→}

   142→// ─── Admin Notification System ──────────────────────────────────────────────
   143→interface AdminNotification {
   144→  id: string
   145→  message: string
   146→  type: 'success' | 'error' | 'info'
   147→  timestamp: Date
   148→}
   149→
   150→const MAX_NOTIFICATIONS = 20
   151→
   152→// Module-level notification store (shared across components)
   153→let _notifications: AdminNotification[] = []
   154→let _listeners: Set<() => void> = new Set()
   155→
   156→function addAdminNotification(message: string, type: AdminNotification['type'] = 'info') {
   157→  const notification: AdminNotification = {
   158→    id: Date.now().toString(36) + Math.random().toString(36).slice(2, 6),
   159→    message,
   160→    type,
   161→    timestamp: new Date(),
   162→  }
   163→  _notifications = [notification, ..._notifications].slice(0, MAX_NOTIFICATIONS)
   164→  _listeners.forEach(l => l())
   165→}
   166→
   167→function getAdminNotifications(): AdminNotification[] {
   168→  return _notifications
   169→}
   170→
   171→function subscribeToNotifications(listener: () => void) {
   172→  _listeners.add(listener)
   173→  return () => _listeners.delete(listener)
   174→}

   513→// ─── Panel General View ─────────────────────────────────────────────────────
   514→function PanelGeneralView() {
   515→  const [stats, setStats] = useState<Stats | null>(null)
   516→  const [loading, setLoading] = useState(true)
   517→  const [error, setError] = useState<string | null>(null)
   518→  const { toast } = useToast()
   519→
   520→  const fetchStats = useCallback(async () => {
   521→    try {
   522→      setLoading(true)
   523→      setError(null)
   524→      const res = await fetch('/api/admin/stats')
   525→      if (!res.ok) throw new Error('Error al cargar estadísticas')
   526→      const data = await res.json()
   527→      setStats(data)
   528→    } catch (err) {
   529→      setError(err instanceof Error ? err.message : 'Error desconocido')
   530→      toast({ title: 'Error', description: 'No se pudieron cargar las estadísticas', variant: 'destructive' })
   531→    } finally {
   532→      setLoading(false)
   533→    }
   534→  }, [toast])
   535→
   536→  useEffect(() => { fetchStats() }, [fetchStats])
   537→
   538→  if (loading) {
   539→    return (
   540→      <div className="space-y-6">
   541→        <div>
