   624→  const dividends: DashboardDividend[] = []
   625→  const investmentConfigs = [
   626→    { assetId: 'centro-logistico-stgo-norte', investmentId: 'inv_001', assetName: 'Centro Logístico Santiago Norte', quantity: 10, pricePerFraction: 250000, monthlyRent: 19800000, totalFractions: 11200 },

   629→  ]
   630→
   631→  let divCounter = 0
   632→  for (const cfg of investmentConfigs) {
   633→    const perFraction = Math.round((cfg.monthlyRent / cfg.totalFractions) * 0.9)
   634→    const monthlyDiv = Math.round(perFraction * cfg.quantity)
   635→    const completedAt = new Date(`2024-01-15T00:00:00.000Z`)
   636→
   637→    for (let m = 0; m < 6; m++) {
   638→      divCounter++
   639→      const periodStart = new Date(2024, m, 1)
   640→      const periodEnd = new Date(2024, m + 1, 0)
   641→      const paymentDate = new Date(2024, m + 1, 5)
   642→      dividends.push({
   643→        id: `div_${String(divCounter).padStart(3, '0')}`,
   644→        amount: monthlyDiv,
   645→        perFraction,
   646→        fractions: cfg.quantity,
   647→        periodStart: periodStart.toISOString(),
   648→        periodEnd: periodEnd.toISOString(),
   649→        paymentDate: paymentDate.toISOString(),
   650→        status: 'paid',
   651→        createdAt: paymentDate.toISOString(),
   652→        investment: {
   653→          id: cfg.investmentId,
   654→          asset: {
   655→            id: cfg.assetId,
   656→            name: cfg.assetName,
   657→          },
   658→        },
   659→      })
   660→    }
   661→  }
   662→
   663→  return dividends
