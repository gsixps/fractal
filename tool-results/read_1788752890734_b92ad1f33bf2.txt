    82→// ─── Helpers ──────────────────────────────────────────────────────────
    83→
    84→function getTimeRemaining(expiresAt: string | null): string {
    85→  if (!expiresAt) return ''
    86→  const now = new Date()
    87→  const expiry = new Date(expiresAt)
    88→  const diffMs = expiry.getTime() - now.getTime()
    89→
    90→  if (diffMs <= 0) return 'Expirada'
    91→
    92→  const days = Math.floor(diffMs / (1000 * 60 * 60 * 24))
    93→  if (days > 0) return `${days}d restantes`
    94→  const hours = Math.floor(diffMs / (1000 * 60 * 60))
    95→  if (hours > 0) return `${hours}h restantes`
    96→  const minutes = Math.floor(diffMs / (1000 * 60))
    97→  return `${minutes}m restantes`
    98→}

   106→    case 'cancelled':
   107→      return <Badge className="bg-gray-100 text-gray-600 border-gray-200 hover:bg-gray-100">Cancelada</Badge>
   108→    case 'partial':
   109→      return <Badge className="bg-amber-100 text-amber-700 border-amber-200 hover:bg-amber-100">Parcial</Badge>
   110→    default:
   111→      return <Badge variant="outline">{status}</Badge>
   112→  }
